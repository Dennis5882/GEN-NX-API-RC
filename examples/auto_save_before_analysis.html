MIDAS Gen NX MAPI — Auto Save Before Analysis

해석 실행(POST /doc/anal) 전에 파일을 자동 저장하는 패턴입니다.
GET /mapikey/verify에서 로그인한 사용자 이메일을 가져와 저장 경로를 동적으로 생성합니다.

동작 흐름

// 1. /mapikey/verify → 사용자 이메일 확인
// 2. 이메일 @ 앞 부분(username) 추출 → Windows 경로 구성
// 3. 타임스탬프로 파일명 유니크화
// 4. POST /doc/saveas → 저장 완료 후
// 5. POST /doc/anal  → 해석 실행

핵심 코드

/**
 * Gen NX MAPI Base URL (예: http://127.0.0.1:8090/gen)
 * /gen, /civil 경로를 제거해 루트 URL 반환 → /mapikey/verify 호출에 사용
 */
function mapiRoot(base) {
  return base.replace(/\/(gen|civil)\/?$/i, '');
}

/**
 * 연결 확인 + 로그인 사용자 정보 반환
 * j.user 에 로그인 이메일이 담겨 있음 (예: "sjj0507@midasit.com")
 */
async function verifyConn(base, key) {
  const res = await fetch(`${mapiRoot(base)}/mapikey/verify`, {
    headers: { 'MAPI-Key': key }
  });
  let j = {};
  try { j = await res.json(); } catch (e) {}
  return { ok: res.ok, j };
}

/**
 * 저장 경로 자동 결정
 *
 *  성공 경로: C:/Users/{username}/Documents/gen_rc_temp_{timestamp}.mgbx
 *  폴백 경로: C:/Temp/gen_rc_temp_{timestamp}.mgbx
 *
 * MAPI 서버는 Gen NX가 실행 중인 PC에서 동작하므로
 * 경로 역시 그 PC 기준으로 구성해야 합니다.
 * %USERPROFILE% 등 환경변수는 MAPI가 인식하지 못하므로
 * /mapikey/verify 의 j.user 이메일로 username 을 직접 추출합니다.
 */
async function autoSavePath(base, key) {
  try {
    const { ok, j } = await verifyConn(base, key);
    if (ok && j && j.user) {
      const username = j.user.split('@')[0];
      return `C:/Users/${username}/Documents/gen_rc_temp_${Date.now()}.mgbx`;
    }
  } catch (e) {}
  // verifyConn 실패 시 공용 임시 폴더로 폴백
  return `C:/Temp/gen_rc_temp_${Date.now()}.mgbx`;
}

/**
 * 해석 실행
 *
 * ⚠️  순서가 중요합니다.
 *    /doc/saveas 먼저 → 그 다음 /doc/anal
 *    순서가 바뀌면 Gen NX가 저장 다이얼로그를 띄울 수 있습니다.
 */
async function runAnalysis(base, key) {
  const savePath = await autoSavePath(base, key);

  // 1) 저장
  await fetch(`${base}/doc/saveas`, {
    method: 'POST',
    headers: { 'MAPI-Key': key, 'Content-Type': 'application/json' },
    body: JSON.stringify({ Argument: savePath })
  });

  // 2) 해석
  await fetch(`${base}/doc/anal`, {
    method: 'POST',
    headers: { 'MAPI-Key': key, 'Content-Type': 'application/json' },
    body: JSON.stringify({})
  });

  return savePath;
}
